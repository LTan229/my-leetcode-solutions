import heapq

class MedianFinder:

    def __init__(self):
        self.large = [] # min heap
        self.small = [] # max heap

    def addNum(self, num: int) -> None:
        # push
        if not self.small:
            heapq.heappush(self.large, num)
        elif num < self.small[0]:
            heapq.heappush_max(self.small, num)
        else:
            heapq.heappush(self.large, num)
        # balance
        while len(self.small) > len(self.large):
            popped = heapq.heappop_max(self.small)
            heapq.heappush(self.large, popped)
        while len(self.small) < len(self.large) - 1:
            popped = heapq.heappop(self.large)
            heapq.heappush_max(self.small, popped)

    def findMedian(self) -> float:
        if len(self.small) == len(self.large):
            return (self.small[0] + self.large[0]) / 2
        return self.large[0]
