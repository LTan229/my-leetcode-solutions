class Solution:
    def maxProduct(self, nums: List[int]) -> int:
        n = len(nums)

        if n == 1:
            return nums[0]

        result = nums[0]
        maxi = mini = result

        for i in range(1, n):
            if nums[i] < 0:
                maxi, mini = mini, maxi

            maxi = max(nums[i], maxi * nums[i])
            mini = min(nums[i], mini * nums[i])

            result = max(result, maxi)
            print(maxi, mini)
        
        return result
