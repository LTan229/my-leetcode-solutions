# DP
```python
class Solution:
    def longestValidParentheses(self, s: str) -> int:
        if len(s) < 2:
            return 0
        dp = [0] * len(s) # maximum length of valid string ending at position i
        result = 0
        for i in range(1, len(s)):
            if s[i] == '(':
                continue
            # s[i] == ')'
            if s[i - 1] == '(':
                dp[i] = dp[i - 2] + 2
            else: # '))'
                if i - dp[i - 1] - 1 >= 0 and s[i - dp[i - 1] - 1] == '(': # if current ) has a consecutive match right before the sequence ending at dp[i - 1]
                    dp[i] = dp[i - 1] + 2 + dp[i - dp[i - 1] - 2] # if dp[i - dp[i - 1] - 2] is ) then add the valid sequence ending with it. if it is (, then consecutive valid sequence is broken, add 0
            result = max(result, dp[i])
        print(dp)
        return result
```

# 栈

```python
class Solution:
    def longestValidParentheses(self, s: str) -> int:
        stack = [-1] # starting point
        result = 0
        for i, char in enumerate(s):
            if char == '(':
                stack.append(i)
            else:
                stack.pop() # pops the index of a matching ( or the last starting point

                if not stack: # popped is starting point
                    stack.append(i) # append cur as the new starting point
                else: # popped is a matching (
                    result = max(result, i - stack[-1]) # updates the max length. current length is the diff between current index and the remaining starting point in the stack
        return result

```

# 双指针

```python
class Solution:
    def longestValidParentheses(self, s: str) -> int:
        left = right = max_len = 0
        
        # left to right
        for char in s:
            if char == '(':
                left += 1
            else:
                right += 1
            
            if left == right: # end of a complete sequence
                max_len = max(max_len, 2 * right)
            elif right > left: # breaks a sequence
                left = right = 0 # reset

        left = right = 0
        
        # right to left
        # if there is always more left then right, maxlen will not be updated
        for char in reversed(s):
            if char == '(':
                left += 1
            else:
                right += 1
                
            if left == right:
                max_len = max(max_len, 2 * left)
            elif left > right:
                left = right = 0
                
        return max_len
```
