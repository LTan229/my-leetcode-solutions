# 堆

```python
import heapq
class Solution:
    def findKthLargest(self, nums: List[int], k: int) -> int:
        heapq.heapify_max(nums)
        for i in range(k):
            cur = heapq.heappop_max(nums)
        
        return cur
```

# 桶
```python
import heapq
class Solution:
    def findKthLargest(self, nums: List[int], k: int) -> int:
        mini = min(nums)
        maxi = max(nums)
        bins = [0] * (maxi - mini + 1)
        for i in nums:
            bins[i - mini] += 1
        
        for i in range(len(bins) - 1, -1, -1):
            k -= bins[i]
            if k <= 0:
                return i + mini

```

# 快排
超时
```python
import random

class Solution:
    def findKthLargest(self, nums: List[int], k: int) -> int:
        target = len(nums) - k

        def select(left, right):
            pivot_idx = random.randint(left, right)
            nums[pivot_idx], nums[right] = nums[right], nums[pivot_idx]

            pivot = nums[right]
            slow = left # left of slow is a sequence that is smaller than pivot
            for fast in range(left, right):
                if nums[fast] < pivot:
                    nums[slow], nums[fast] = nums[fast], nums[slow]
                    slow += 1

            nums[slow], nums[right] = nums[right], nums[slow]

            if slow == target:
                return nums[slow]
            if slow < target:
                return select(slow + 1, right)
            else:
                return select(left, slow - 1)
        
        return select(0, len(nums) - 1)

```
