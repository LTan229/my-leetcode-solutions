我一开始思路是……两个指针分别指着两个数组开头，然后把更小的指针往后挪。
```python
class Solution:
    def findMedianSortedArrays(self, nums1: List[int], nums2: List[int]) -> float:

        def getKthElement(k):
            index1, index2 = 0, 0
            while True:
                if index1 == len1:
                    return nums2[index2 + k - 1]
                if index2 == len2:
                    return nums1[index1 + k - 1]
                if k == 1:
                    return min(nums1[index1], nums2[index2])

                # look for k/2 th smallest in each array
                newIndex1 = min(index1 + k // 2 - 1, len1 - 1)
                newIndex2 = min(index2 + k // 2 - 1, len2 - 1)
                pivot1, pivot2 = nums1[newIndex1], nums2[newIndex2]
                
                if pivot1 <= pivot2:
                    # k/2-1 elements in nums1 and at most k/2-1 elements in nums2 is less than pivot1
                    # so it is at most k-1 th smallest, safe to rule it out
                    # pivot1 and all prev items excluded
                    k -= (newIndex1 - index1 + 1) 
                    index1 = newIndex1 + 1
                else:
                    # same as above
                    k -= (newIndex2 - index2 + 1)
                    index2 = newIndex2 + 1

        len1, len2 = len(nums1), len(nums2)
        tot_len = len1 + len2
        if tot_len % 2 == 1:
            return getKthElement(tot_len // 2 + 1)
        else:
            return (getKthElement(tot_len // 2) + getKthElement(tot_len // 2 + 1)) / 2

```
