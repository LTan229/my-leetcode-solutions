# 做差
```python
# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, x):
#         self.val = x
#         self.next = None

class Solution:
    def getIntersectionNode(self, headA: ListNode, headB: ListNode) -> Optional[ListNode]:
        lenA, lenB = 0, 0
        
        curA = headA
        while curA:
            lenA += 1
            curA = curA.next
        
        curB = headB
        while curB:
            lenB += 1
            curB = curB.next

        lendiff = lenA - lenB
        if lendiff > 0:
            for _ in range(lendiff):
                headA = headA.next
        else:
            for _ in range(0 - lendiff):
                headB = headB.next
                lendiff += 1

        lenrem = min(lenA, lenB)
        for _ in range(lenrem):
            if headA == headB:
                return headA
            headA = headA.next
            headB = headB.next
        return None
        
```
# 交叉指针
```python
# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, x):
#         self.val = x
#         self.next = None

class Solution:
    def getIntersectionNode(self, headA: ListNode, headB: ListNode) -> Optional[ListNode]:
        if not headA or not headB:
            return None
        pa, pb = headA, headB
        while pa != pb:
            pa = pa.next if pa else headB # walk through A, None, then B. None serves at the intersection when A and B are not intersected
            pb = pb.next if pb else headA
        
        return pa
```
