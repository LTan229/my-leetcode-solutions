```python
class Node:

    def __init__(self, 
                key=0,
                val=0, 
                nxt=None,
                prev=None
                ):
        self.key = key
        self.val = val
        self.next = nxt
        self.prev = prev    

class LRUCache:

    def __init__(self, capacity: int):
        self.d = {} # O1 get
        self.len = 0
        self.capacity = capacity

        self.head = Node()
        self.tail = Node(prev=self.head)
        self.head.next = self.tail

    def stringify(self):
        cur = self.head.next
        result = ""
        while cur != self.tail:
            result = result + str(cur.val) + ", "
            cur = cur.next
        print(result[:-2])

    def get(self, key: int) -> int:
        if key not in self.d:
            return -1
        self.move2head(key)
        # self.stringify()
        return self.d[key].val

    def put(self, key: int, value: int) -> None:
        if key in self.d: # modify count as "visited"?
            # modify
            self.d[key].val = value
            # move to head
            self.move2head(key)
        else:
            # insert
            tmp = self.head.next 
            self.d[key] = Node(key, value, nxt=tmp, prev=self.head)
            self.head.next = self.d[key]
            tmp.prev = self.d[key]
            self.len += 1
            # check len
            if self.len > self.capacity:
                pop_node = self.tail.prev
                prev = self.tail.prev.prev

                self.d.pop(pop_node.key)

                prev.next = self.tail
                self.tail.prev = prev

                self.len -= 1
        # self.stringify()

    def move2head(self, key):
        # remove
        prev = self.d[key].prev
        nxt = self.d[key].next
        prev.next = nxt
        nxt.prev = prev
        # insert to head
        first = self.head.next
        self.head.next = self.d[key]
        self.d[key].prev = self.head
        self.d[key].next = first
        first.prev = self.d[key]
```

```python
class Node:

    def __init__(self, 
                key=0,
                val=0, 
                nxt=None,
                prev=None
                ):
        self.key = key
        self.val = val
        self.next = nxt
        self.prev = prev    

class LRUCache:

    def __init__(self, capacity: int):
        self.d = {} # O1 get
        self.capacity = capacity

        self.head = Node() # double linked list: O1 insert and pop
        self.tail = Node(prev=self.head)
        self.head.next = self.tail

    def get(self, key: int) -> int:
        if key not in self.d:
            return -1
        self._move_to_head(self.d[key])
        return self.d[key].val

    def put(self, key: int, value: int) -> None:
        if key in self.d: # modify count as "visited"
            self.d[key].val = value
            self._move_to_head(self.d[key])
        else:
            # insert
            new_node = Node(key, value)
            self.d[key] = new_node
            self._add_to_head(new_node)
            # check len
            if len(self.d) > self.capacity:
                pop_node = self.tail.prev
                self._remove_node(pop_node)
                self.d.pop(pop_node.key)

    def _remove_node(self, node):
        node.prev.next = node.next
        node.next.prev = node.prev

    def _add_to_head(self, node):
        node.next = self.head.next
        node.prev = self.head
        self.head.next.prev = node
        self.head.next = node

    def _move_to_head(self, node):
        self._remove_node(node)
        self._add_to_head(node)
```
