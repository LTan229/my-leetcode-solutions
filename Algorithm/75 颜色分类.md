可以使用 三指针（左、右、当前） 的方法，一次性把 0 扔到左边，2 扔到右边。
```python
class Solution:
    def sortColors(self, nums: List[int]) -> None:
        fast = slow = 0
        while fast < len(nums):
            if nums[fast] == 0:
                nums[fast], nums[slow] = nums[slow], nums[fast]
                slow += 1
            fast += 1
        
        fast = slow = len(nums) - 1
        while fast >= 0:
            if nums[fast] == 2:
                nums[fast], nums[slow] = nums[slow], nums[fast]
                slow -= 1
            fast -= 1

```
