```python
from collections import deque

class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        n = len(grid)
        m = len(grid[0])
        dirs = [[1, 0], [-1, 0], [0, 1], [0, -1]]
        count = 0

        def ingrid(x, y):
            return 0 <= x < n and 0 <= y < m

        for i in range(n):
            for j in range(m):
                if grid[i][j] == '0':
                    continue
                count += 1
                stack = deque([(i, j)])
                grid[i][j] = '0'
                while stack:
                    cur = stack.popleft()
                    for d in dirs:
                        newi = cur[0] + d[0]
                        newj = cur[1] + d[1]
                        if ingrid(newi, newj) and grid[newi][newj] == '1':
                            grid[newi][newj] = '0'
                            stack.append((newi, newj))

        return count
```
