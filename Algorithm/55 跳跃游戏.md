```python
class Solution:
    def canJump(self, nums: List[int]) -> bool:
        maxdist = 0
        for i in range(len(nums)):
            if i > maxdist:
                break
            maxdist = max(i + nums[i], maxdist)
        if maxdist >= len(nums) - 1:
            return True
        return False
```
