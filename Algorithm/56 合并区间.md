```Python
class Solution:
    def merge(self, intervals: List[List[int]]) -> List[List[int]]:
        intervals = sorted(intervals)
        results = []
        for start, end in intervals:
            if not results:
                results.append([start, end])
                continue
            if start <= results[-1][1]:
                results[-1][1] = max(end, results[-1][1]) # if the comming interval is inclued in the original raenge
            else:
                results.append([start, end])
        return results
```
