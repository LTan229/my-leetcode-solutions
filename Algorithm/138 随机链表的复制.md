# 哈希表
```python
"""
# Definition for a Node.
class Node:
    def __init__(self, x: int, next: 'Node' = None, random: 'Node' = None):
        self.val = int(x)
        self.next = next
        self.random = random
"""

class Solution:
    def copyRandomList(self, head: 'Optional[Node]') -> 'Optional[Node]':
        if not head:
            return head
        d = {}
        i = 0
        cur = head
        lst = []
        while cur:
            # record index
            d[cur] = i

            # create copy
            cpy = Node(cur.val)
            if lst:
                lst[-1].next = cpy
            lst.append(cpy)

            # increment
            i += 1
            cur = cur.next
        
        cur = head
        i = 0
        while cur:
            if cur.random: # is not none
                rand_idx = d[cur.random]
                lst[i].random = lst[rand_idx]
            i += 1
            cur = cur.next
        
        return lst[0]
        

```

# 不用额外空间
```python
class Solution:
    def copyRandomList(self, head: 'Optional[Node]') -> 'Optional[Node]':
        if not head:
            return None
        
        cur = head
        while cur: # after each node insert its copy
            new_node = Node(cur.val, cur.next)
            cur.next = new_node
            cur = new_node.next
            
        cur = head
        while cur: # set random
            if cur.random:
                # copy.random <- random.copy
                cur.next.random = cur.random.next
            cur = cur.next.next
            
        cur = head
        new_head = head.next
        while cur:
            clone = cur.next
            cur.next = clone.next
            if clone.next: # if has next element
                clone.next = clone.next.next
            cur = cur.next
            
        return new_head
```
