```python
class Solution:
    def jump(self, nums: List[int]) -> int:
        if len(nums) <= 1:
            return 0
        
        jumps = 0
        cur_end = 0
        next_end = 0

        for i in range(len(nums)):
            next_end = max(next_end, i + nums[i])
            if next_end >= len(nums) - 1:
                return jumps + 1
            if i == cur_end:
                cur_end = next_end
                jumps += 1
        
```
