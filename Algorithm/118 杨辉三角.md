```python
class Solution:
    def generate(self, numRows: int) -> List[List[int]]:
        result = [[1]]
        
        if numRows == 1:
            return result
        
        for rowidx in range(1, numRows):
            row = [1]
            for colidx in range(1, rowidx):
                row.append(result[-1][colidx - 1] + result[-1][colidx])
            row.append(1)
            result.append(row)
        return result
```
