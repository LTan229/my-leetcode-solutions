# 栈
```python
# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def reverseKGroup(self, head: Optional[ListNode], k: int) -> Optional[ListNode]:
        if not head or not head.next:
            return head
        dummy = ListNode(0, head)
        prv, nxt = dummy, head
        while nxt:
            stack = []
            i = 0
            while nxt and i < k:
                stack.append(nxt)
                nxt = nxt.next
                i += 1
            if i < k: # not a whole group, quit
                break 
            for i in range(k - 1, -1, -1):
                stack[i].next = stack[i - 1]
            prv.next = stack[-1]
            stack[0].next = nxt
            prv = stack[0]
        return dummy.next
```

# 原地翻转

```python
# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def reverseKGroup(self, head: Optional[ListNode], k: int) -> Optional[ListNode]:
        if not head or k == 1:
            return head
        dummy = ListNode(0, head)
        group_prv = dummy

        while True:
            kth = self.getNextKth(group_prv, k)
            if not kth:
                break
            group_nxt = kth.next

            prev, cur = group_nxt, group_prv.next
            while cur != group_nxt:
                nxt = cur.next
                cur.next = prev
                prev = cur
                cur = nxt

            tmp = group_prv.next 
            group_prv.next = prev 
            group_prv = tmp
            
        return dummy.next

    def getNextKth(self, cur, k):
        while cur and k > 0:
            cur = cur.next
            k -= 1
        return cur
```
