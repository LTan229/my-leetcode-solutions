class Solution {
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
        int m = matrix.size();
        int n = matrix[0].size();
        vector<vector<int>> visited(m, vector<int>(n, 0));
        // vector<vector<int>> next = {{0,1},{1,0},{0,-1},{-1,0}};
        int i = 0;
        int j = 0;
        while (i >= 0 && i < m && j >= 0 && j < n) {
            // if greater than current line range, go down
            if (target > matrix[i][n - 1]) {
                i += 1;
                continue;
            }
            // if smaller than urrent line range, quit
            if (target < matrix[i][0]) {
                break;
            }
            // in cur line range
            // if equal, return true
            if (target == matrix[i][j]) {
                return true;
            }
            // if greater than current element, go right
            if (target > matrix[i][j]) {
                j += 1;
                continue;
            }
            // if smaller than current ele, quit
            if (target < matrix[i][j]) {
                break;
            }
        }
        return false;
    }
};

class Solution {
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
        if (matrix.empty() || matrix[0].empty()) return false;
        
        int m = matrix.size();
        int n = matrix[0].size();
        
        // 从右上角出发
        int i = 0, j = n - 1;
        
        while (i < m && j >= 0) {
            if (matrix[i][j] == target) {
                return true;
            } else if (matrix[i][j] > target) {
                j--; // 这一列都太大了，往左移
            } else {
                i++; // 这一行都太小了，往下移
            }
        }
        return false;
    }
};
