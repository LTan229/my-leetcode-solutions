class Solution:
    def longestPalindrome(self, s: str) -> str:
        n = len(s)
        if n < 2:
            return s
        dp = [[0] * n for _ in range(n)]
        maxlen = 0
        maxtuple = (0, 0)
        for start in range(n - 1, -1, -1):
            for end in range(start, n):
                # print(start, end)
                if s[start] == s[end]:
                    # print(s[start], s[end])
                    if end - start < 3:
                        dp[start][end] = end - start + 1
                    else:
                        if dp[start + 1][end - 1] == 0:
                            dp[start][end] = 0
                        else:
                            dp[start][end] = dp[start + 1][end - 1] + 2
                if dp[start][end] > maxlen:
                    maxlen = dp[start][end]
                    maxtuple = (start, end)
        # print(dp)
        return s[maxtuple[0]:maxtuple[1] + 1]
