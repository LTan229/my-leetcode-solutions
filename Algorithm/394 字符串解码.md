class Solution:
    def decodeString(self, s: str) -> str:
        stack = []
        for char in s:
            if char == "]":
                local_str = []
                while True:
                    temp = stack.pop()
                    if temp == "[":
                        break
                    else:
                        local_str.append(temp)
                repeat = []
                while stack and stack[-1].isdigit():
                    repeat.append(stack.pop())
                repeat = int("".join(repeat[::-1]))
                stack += local_str[::-1] * repeat
            else:
                stack.append(char)
        return "".join(stack)
