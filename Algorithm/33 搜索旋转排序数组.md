宗旨是判断 target 在 mid 哪边

```python
class Solution:
    def search(self, nums: List[int], target: int) -> int:
        if target == nums[0]:
            return 0

        left, right = 0, len(nums) - 1

        while left <= right:

            mid = left + (right - left) // 2
            print(left, mid, right)
            if target == nums[mid]:
                return mid
            
            if nums[left] <= nums[mid]: # left ordered
                if nums[left] <= target < nums[mid]: # target at left
                    right = mid - 1
                else: # target at right
                    left = mid + 1
            else: # right ordered
                if nums[mid] < target <= nums[right]: # target at right
                    left = mid + 1
                else: # target at left
                    right = mid - 1

        return -1
```
