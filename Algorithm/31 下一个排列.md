class Solution:
    def nextPermutation(self, nums: List[int]) -> None:
        """
        Do not return anything, modify nums in-place instead.
        """
        if len(nums) == 1:
            return nums

        i = len(nums) - 1
        while i > 0 and nums[i-1] >= nums[i]:
            i -= 1

        # nums[i - 1]  is the number to replace. swwap it with the next number greater than nums[i - 1] in the right part
        if i > 0:
            j = len(nums) - 1
            while nums[j] <= nums[i - 1]:
                j -= 1
            nums[i-1], nums[j] = nums[j], nums[i-1]
        
        # reverse the descending part to make it as small as possible
        left, right = i, len(nums) - 1
        while left < right:
            nums[left], nums[right] = nums[right], nums[left]
            left += 1
            right -= 1
