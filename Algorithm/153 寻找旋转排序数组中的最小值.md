        minimal = nums[0]

class Solution:
    def findMin(self, nums: List[int]) -> int:
        left, right = 0, len(nums) - 1
        while left < right: # exits when[left, right] only contains target
            mid = left + (right - left) // 2
            if nums[mid] > nums[right]: # judged on the rhs because left ordered may imply a rotation step of 0
                left = mid + 1
            else:
                right = mid
        
        return nums[left]
